# python-dotenv   : loads secrets from .env into config.py

#   BigQuery   -> sqlalchemy-bigquery
#   Redshift   -> sqlalchemy-redshift   (plus redshift-connector or psycopg2)
#   Databricks -> databricks-sqlalchemy
#   MySQL      -> pymysql
